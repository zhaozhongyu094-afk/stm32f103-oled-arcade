cmake_minimum_required(VERSION 3.22)
project(arcade_native_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
foreach(game snake runner post)
    add_executable(test_${game} test_${game}.c ../Display/gfx.c)
    target_include_directories(test_${game} PRIVATE ../Common ../Display)
    if(MSVC)
        target_compile_options(test_${game} PRIVATE /UNDEBUG)
    else()
        target_compile_options(test_${game} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${game} COMMAND test_${game})
endforeach()
add_executable(test_buttons test_buttons.c)
target_include_directories(test_buttons PRIVATE fakes ../BSP ../Common)
if(MSVC)
    target_compile_options(test_buttons PRIVATE /UNDEBUG)
else()
    target_compile_options(test_buttons PRIVATE -UNDEBUG)
endif()
add_test(NAME buttons COMMAND test_buttons)
add_executable(test_scheduler test_scheduler.c ../Display/gfx.c)
target_include_directories(test_scheduler PRIVATE fakes ../BSP ../Common ../App ../Display)
if(MSVC)
    target_compile_options(test_scheduler PRIVATE /UNDEBUG)
else()
    target_compile_options(test_scheduler PRIVATE -UNDEBUG)
endif()
add_test(NAME scheduler COMMAND test_scheduler)
